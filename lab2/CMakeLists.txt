cmake_minimum_required(VERSION 3.16)
project(OSLab2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(process_library STATIC src/ProcessLibrary.cpp)
target_include_directories(process_library PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(child_program tests/child.cpp)
add_executable(process_test tests/main.cpp)
target_link_libraries(process_test PRIVATE process_library)
enable_testing()
add_test(NAME process_library_tests COMMAND process_test $<TARGET_FILE:child_program>)
